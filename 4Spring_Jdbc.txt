Step 4: AppConfig
package config;

import javax.sql.DataSource;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import org.springframework.jdbc.core.JdbcTemplate;

import org.springframework.jdbc.datasource.DriverManagerDataSource;

@Configuration
public class AppConfig {

    @Bean
    public DataSource dataSource() {

        DriverManagerDataSource ds = new DriverManagerDataSource();

        ds.setDriverClassName("com.mysql.cj.jdbc.Driver");

        ds.setUrl("jdbc:mysql://localhost:3306/springjdbc");

        ds.setUsername("root");

        ds.setPassword("root");

        return ds;
    }

    @Bean
    public JdbcTemplate jdbcTemplate() {

        return new JdbcTemplate(dataSource());

    }

}

Step 6: DAO Implementation
package dao;

import config.AppConfig;
import entity.Student;

import java.util.List;

import org.springframework.context.annotation.AnnotationConfigApplicationContext;

import org.springframework.jdbc.core.BeanPropertyRowMapper;

import org.springframework.jdbc.core.JdbcTemplate;

public class StudentDaoImpl implements StudentDao {

    AnnotationConfigApplicationContext context =
            new AnnotationConfigApplicationContext(AppConfig.class);

    JdbcTemplate jdbcTemplate = context.getBean(JdbcTemplate.class);

    // INSERT
    @Override
    public int insert(Student student) {

        String sql =
                "insert into student values(?,?,?,?)";

        return jdbcTemplate.update(
                sql,
                student.getId(),
                student.getName(),
                student.getAge(),
                student.getCourse()
        );

    }

    // UPDATE
    @Override
    public int update(Student student) {

        String sql =
                "update student set name=?,age=?,course=? where id=?";

        return jdbcTemplate.update(
                sql,
                student.getName(),
                student.getAge(),
                student.getCourse(),
                student.getId()
        );

    }

    // DELETE
    @Override
    public int delete(int id) {

        String sql =
                "delete from student where id=?";

        return jdbcTemplate.update(sql,id);

    }

    // SELECT ONE
    @Override
    public Student getStudent(int id) {

        String sql =
                "select * from student where id=?";

        return jdbcTemplate.queryForObject(
                sql,
                new BeanPropertyRowMapper<>(Student.class),
                id
        );

    }

    // SELECT ALL
    @Override
    public List<Student> getAllStudents() {

        String sql =
                "select * from student";

        return jdbcTemplate.query(
                sql,
                new BeanPropertyRowMapper<>(Student.class)
        );

    }

}

import dao.StudentDao;

import dao.StudentDaoImpl;

import entity.Student;

public class Main {

        StudentDao dao = new StudentDaoImpl();

        // UPDATE

        System.out.println(dao.update(s));



        // GET ONE

Q: Why do we pass Student.class to BeanPropertyRowMapper?

Answer: BeanPropertyRowMapper uses Student.class to create a Student object using reflection. It then matches database column names (like id, name, age, course) with the corresponding setter methods (setId(), setName(), etc.) and populates the object automatically.
