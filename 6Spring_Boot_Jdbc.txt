public class Student {

Step 5: Repository Interface
import java.util.List;

public interface StudentRepository {

    int save(Student student);

    int update(Student student);

    int delete(int id);

    Student getStudent(int id);

    List<Student> getAllStudents();

}
Step 6: Repository Implementation
@Repository
public class StudentRepositoryImpl
        implements StudentRepository {

    @Autowired
    private JdbcTemplate jdbcTemplate;

    // INSERT
    @Override
    public int save(Student student) {

        String sql =
        "insert into student(name,email,course,marks) values(?,?,?,?)";

        return jdbcTemplate.update(
                sql,
                student.getName(),
                student.getEmail(),
                student.getCourse(),
                student.getMarks()
        );
    }

    // UPDATE
    @Override
    public int update(Student student) {

        String sql =
        "update student set name=?,email=?,course=?,marks=? where id=?";

        return jdbcTemplate.update(
                sql,
                student.getName(),
                student.getEmail(),
                student.getCourse(),
                student.getMarks(),
                student.getId()
        );
    }

    // DELETE
    @Override
    public int delete(int id) {

        String sql =
        "delete from student where id=?";

        return jdbcTemplate.update(sql,id);
    }

    // SELECT BY ID
    @Override
    public Student getStudent(int id) {

        String sql =
        "select * from student where id=?";

        return jdbcTemplate.queryForObject(
                sql,
                new BeanPropertyRowMapper<>(Student.class),
                id
        );

    }

    // SELECT ALL
    @Override
    public List<Student> getAllStudents() {

        String sql =
        "select * from student";

        return jdbcTemplate.query(
                sql,
                new BeanPropertyRowMapper<>(Student.class)
        );

    }

}
Step 7: Service Interface
import java.util.List;

public interface StudentService {

    int save(Student student);

    int update(Student student);

    int delete(int id);

    List<Student> getAllStudents();

Step 8: Service Implementation
@Service
public class StudentServiceImpl
        implements StudentService {

    @Autowired
    private StudentRepository repository;

    @Override
    public int save(Student student) {
        return repository.save(student);
    }

    @Override
    public int update(Student student) {
        return repository.update(student);
    }

    @Override
    public int delete(int id) {
        return repository.delete(id);
    }

    @Override
    public Student getStudent(int id) {
        return repository.getStudent(id);
    }

    @Override
    public List<Student> getAllStudents() {
        return repository.getAllStudents();
    }

}
Step 9: Controller
@RestController
@RequestMapping("/students")
public class StudentController {

    @Autowired
    private StudentService service;

    // INSERT
    @PostMapping
    public String save(@RequestBody Student student){

        service.save(student);

        return "Student Saved Successfully";
    }

    // UPDATE
    @PutMapping
    public String update(@RequestBody Student student){

        service.update(student);

        return "Student Updated Successfully";
    }

    // DELETE
    @DeleteMapping("/{id}")
    public String delete(@PathVariable int id){

        service.delete(id);

        return "Student Deleted Successfully";
    }

    // GET BY ID
    @GetMapping("/{id}")
    public Student getStudent(@PathVariable int id){

        return service.getStudent(id);
    }

    // GET ALL
    @GetMapping
    public List<Student> getAllStudents(){

        return service.getAllStudents();
    }

}
